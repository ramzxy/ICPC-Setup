\subsection{Create and run}\label{setup:debug}
\begin{lstlisting}[language=bash]
mkdir -p contest/tmpl
cd contest            # save files as shown
source s.sh           # repeat in each terminal
tmpl                  # copy tmpl/ into A-N
rp A                  # Python, terminal input
rc A                  # compile and run C++
rp A 1                # reads A/1.in
rc A 1
tp A                  # test Python on all A/*.in
tc A                  # compile once, test all C++
\end{lstlisting}
\end{minipage}\par
\hint{Pair \texttt{A/1.in} with expected \texttt{A/1.out}.
Replace the copied samples for each problem.
All cases run; any failure gives a nonzero status. Spacing/blank lines are
ignored; custom or floating-point outputs need manual checks.}
\hint{\texttt{tmpl} keeps existing files. Letters accept either case.
\texttt{rp A sample} reads \texttt{A/sample.in}; no input argument uses stdin.}
